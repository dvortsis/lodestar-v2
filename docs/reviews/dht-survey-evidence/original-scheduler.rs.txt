/// Abstract time provider so runtime scheduling relies strictly on monotonic elapsed time
/// and tests can simulate forward/backward wall-clock jumps across restarts.
pub trait Clock: Send + Sync + 'static {
    fn wall_unix_secs(&self) -> Result<u32, SchedulerError>;
    fn monotonic_elapsed_secs(&self) -> u64;
}

pub struct SystemClock {
    boot_instant: Instant,
}

impl SystemClock {
    pub fn new() -> Self {
        Self {
            boot_instant: Instant::now(),
        }
    }
}

impl Default for SystemClock {
    fn default() -> Self {
        Self::new()
    }
}

impl Clock for SystemClock {
    fn wall_unix_secs(&self) -> Result<u32, SchedulerError> {
        let raw = SystemTime::now()
            .duration_since(UNIX_EPOCH)
            .map_err(|_| SchedulerError::ClockWentBackwards)?
            .as_secs();
        Ok(u32::try_from(raw).unwrap_or(u32::MAX))
    }

    fn monotonic_elapsed_secs(&self) -> u64 {
        self.boot_instant.elapsed().as_secs()
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SchedulerError {
    SqliteError(String),
    DatabaseOverflow { found: usize, max: usize },
    CorruptDatabaseRecord {
        ip_raw: i64,
        port_raw: i64,
        support_raw: i64,
        novel_raw: i64,
        op_seq_raw: i64,
    },
    ZeroRowsAffected {
        addr: SocketAddrV4,
        op_seq: u64,
    },
    WalChannelClosed,
    WalCommitAckDropped,
    InjectedWalFailure,
    StaleSequence { expected: u64, provided: u64 },
    SequenceOverflow,
    InvalidLifecycleTransition,
    ClockWentBackwards,
}

impl fmt::Display for SchedulerError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::SqliteError(msg) => write!(f, "SQLite persistence error: {msg}"),
            Self::DatabaseOverflow { found, max } => write!(
                f,
                "Active DB records ({found}) exceed configured limit ({max})"
            ),
            Self::CorruptDatabaseRecord {
                ip_raw,
                port_raw,
                support_raw,
                novel_raw,
                op_seq_raw,
            } => write!(
                f,
                "Corrupt DB row (ip={ip_raw}, port={port_raw}, support={support_raw}, novel={novel_raw}, op_seq={op_seq_raw})"
            ),
            Self::ZeroRowsAffected { addr, op_seq } => write!(
                f,
                "SQLite upsert affected 0 rows for {addr} at op_seq={op_seq} (stale sequence rejected)"
            ),
            Self::WalChannelClosed => write!(f, "WAL persistence worker channel closed"),
            Self::WalCommitAckDropped => write!(f, "WAL persistence worker dropped commit ack"),
            Self::InjectedWalFailure => write!(f, "Simulated WAL transaction failure"),
            Self::StaleSequence { expected, provided } => write!(
                f,
                "Stale operation sequence (expected={expected}, provided={provided})"
            ),
            Self::SequenceOverflow => write!(f, "Operation sequence counter overflowed i64::MAX"),
            Self::InvalidLifecycleTransition => write!(f, "Invalid endpoint lifecycle transition"),
            Self::ClockWentBackwards => write!(f, "System clock before UNIX_EPOCH"),
        }
    }
}

impl std::error::Error for SchedulerError {}

impl From<rusqlite::Error> for SchedulerError {
    fn from(e: rusqlite::Error) -> Self {
        Self::SqliteError(e.to_string())
    }
}

/// Min-heap entry for `hot_heap` and `expiry_heap` ordered by earliest `due_epoch_sec`.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct ScheduledNode {
    due_epoch_sec: u32,
    addr: SocketAddrV4,
    op_seq: u64,
}

impl Ord for ScheduledNode {
    fn cmp(&self, other: &Self) -> CmpOrdering {
        other
            .due_epoch_sec
            .cmp(&self.due_epoch_sec)
            .then_with(|| other.op_seq.cmp(&self.op_seq))
            .then_with(|| other.addr.ip().cmp(self.addr.ip()))
            .then_with(|| other.addr.port().cmp(&self.addr.port()))
    }
}

impl PartialOrd for ScheduledNode {
    fn partial_cmp(&self, other: &Self) -> Option<CmpOrdering> {
        Some(self.cmp(other))
    }
}

/// Min-heap entry for `IpRateLimiter` ordered by earliest `full_at_mono_sec`.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct IpReclaimEntry {
    full_at_mono_sec: u64,
    ip: Ipv4Addr,
    bucket_gen: u64,
}

impl Ord for IpReclaimEntry {
    fn cmp(&self, other: &Self) -> CmpOrdering {
        other
            .full_at_mono_sec
            .cmp(&self.full_at_mono_sec)
            .then_with(|| other.bucket_gen.cmp(&self.bucket_gen))
            .then_with(|| other.ip.cmp(&self.ip))
    }
}

impl PartialOrd for IpReclaimEntry {
    fn partial_cmp(&self, other: &Self) -> Option<CmpOrdering> {
        Some(self.cmp(other))
    }
}

/// Strictly validates that an IPv4 socket address is a globally routable public unicast endpoint.
pub fn is_valid_public_dht_endpoint(addr: SocketAddrV4) -> bool {
    if addr.port() == 0 {
        return false;
    }
    let [a, b, c, _d] = addr.ip().octets();
    if a == 0 || a == 10 || a == 127 || a >= 224 {
        return false;
    }
    if a == 100 && (b & 0b1100_0000) == 64 {
        return false; // 100.64.0.0/10 (RFC 6598 CGNAT)
    }
    if a == 169 && b == 254 {
        return false; // 169.254.0.0/16 (Link-Local)
    }
    if a == 172 && (16..=31).contains(&b) {
        return false; // 172.16.0.0/12 (RFC 1918 Private)
    }
    if a == 192 && b == 0 && (c == 0 || c == 2) {
        return false; // 192.0.0.0/24 & 192.0.2.0/24 (TEST-NET-1)
    }
    if a == 192 && b == 88 && c == 99 {
        return false; // 192.88.99.0/24 (6to4 Relay Anycast)
    }
    if a == 192 && b == 168 {
        return false; // 192.168.0.0/16 (RFC 1918 Private)
    }
    if a == 198 && (b == 18 || b == 19) {
        return false; // 198.18.0.0/15 (Benchmarking)
    }
    if a == 198 && b == 51 && c == 100 {
        return false; // 198.51.100.0/24 (TEST-NET-2)
    }
    if a == 203 && b == 0 && c == 113 {
        return false; // 203.0.113.0/24 (TEST-NET-3)
    }
    true
}

#[derive(Clone, Copy)]
struct MillitokenBucket {
    millitokens: u32,
    last_refill_mono_sec: u64,
    full_at_mono_sec: u64,
    bucket_gen: u64,
}

struct IpRateLimiter {
    buckets: HashMap<Ipv4Addr, MillitokenBucket>,
    reclaim_heap: BinaryHeap<IpReclaimEntry>,
    max_entries: usize,
    capacity_mt: u32,
    refill_mt_per_sec: u32,
    next_gen: u64,
}

impl IpRateLimiter {
    fn new(max_entries: usize, capacity_mt: u32, refill_mt_per_sec: u32) -> Self {
        let max_entries = max_entries.max(1);
        Self {
            buckets: HashMap::with_capacity(max_entries.min(65_536)),
            reclaim_heap: BinaryHeap::with_capacity(max_entries.min(65_536)),
            max_entries,
            capacity_mt,
            refill_mt_per_sec: refill_mt_per_sec.max(1),
            next_gen: 1,
        }
    }

    fn compute_full_at(
        now_mono: u64,
        remaining_mt: u32,
        capacity_mt: u32,
        refill_mt_per_sec: u32,
    ) -> u64 {
        let deficit = capacity_mt.saturating_sub(remaining_mt);
        if deficit == 0 {
            return now_mono;
        }
        let secs_to_full = u64::from(deficit).div_ceil(u64::from(refill_mt_per_sec));
        now_mono.saturating_add(secs_to_full)
    }

    /// Proactively prunes expired/stale entries and bounds `reclaim_heap` to `<= 2 * max_entries`.
    fn maintain_heap(&mut self, now_mono: u64, max_steps: usize) {
        for _ in 0..max_steps {
            let Some(& entry) = self.reclaim_heap.peek() else {
                break;
            };
            let is_stale = self
                .buckets
                .get(&entry.ip)
                .map(|b| b.bucket_gen != entry.bucket_gen)
                .unwrap_or(true);
            if is_stale {
                self.reclaim_heap.pop();
                continue;
            }
            if entry.full_at_mono_sec <= now_mono {
                self.reclaim_heap.pop();
                self.buckets.remove(&entry.ip);
            } else {
                break;
            }
        }

        let heap_ceiling = self.max_entries.saturating_mul(2).max(16);
        if self.reclaim_heap.len() > heap_ceiling {
            let buckets_ref = &self.buckets;
            let live: Vec<IpReclaimEntry> = self
                .reclaim_heap
                .drain()
                .filter(|e| {
                    buckets_ref
                        .get(&e.ip)
                        .map(|b| b.bucket_gen == e.bucket_gen)
                        .unwrap_or(false)
                })
                .collect();
            self.reclaim_heap.extend(live);
        }
    }

    fn try_consume(&mut self, ip: Ipv4Addr, now_mono: u64, cost_mt: u32) -> bool {
        if cost_mt == 0 || cost_mt > self.capacity_mt {
            return false;
        }

        self.maintain_heap(now_mono, 4);

        if let Some(bucket) = self.buckets.get_mut(&ip) {
            let elapsed = now_mono.saturating_sub(bucket.last_refill_mono_sec);
            let added = elapsed
                .saturating_mul(u64::from(self.refill_mt_per_sec))
                .min(u64::from(self.capacity_mt)) as u32;
            bucket.millitokens = bucket.millitokens.saturating_add(added).min(self.capacity_mt);
            bucket.last_refill_mono_sec = now_mono;

            if bucket.millitokens < cost_mt {
                return false;
            }
            bucket.millitokens -= cost_mt;
            self.next_gen = self.next_gen.saturating_add(1);
            bucket.bucket_gen = self.next_gen;
            bucket.full_at_mono_sec = Self::compute_full_at(
                now_mono,
                bucket.millitokens,
                self.capacity_mt,
                self.refill_mt_per_sec,
            );
            let updated_entry = IpReclaimEntry {
                full_at_mono_sec: bucket.full_at_mono_sec,
                ip,
                bucket_gen: bucket.bucket_gen,
            };
            self.reclaim_heap.push(updated_entry);
            self.maintain_heap(now_mono, 0);
            return true;
        }

        while self.buckets.len() >= self.max_entries {
            let Some(&entry) = self.reclaim_heap.peek() else {
                break;
            };
            let is_stale = self
                .buckets
                .get(&entry.ip)
                .map(|b| b.bucket_gen != entry.bucket_gen)
                .unwrap_or(true);
            if is_stale {
                self.reclaim_heap.pop();
                continue;
            }
            if entry.full_at_mono_sec <= now_mono {
                self.reclaim_heap.pop();
                self.buckets.remove(&entry.ip);
            } else {
                return false;
            }
        }

        let remaining = self.capacity_mt - cost_mt;
        self.next_gen = self.next_gen.saturating_add(1);
        let bucket_gen = self.next_gen;
        let full_at = Self::compute_full_at(
            now_mono,
            remaining,
            self.capacity_mt,
            self.refill_mt_per_sec,
        );
        self.buckets.insert(
            ip,
            MillitokenBucket {
                millitokens: remaining,
                last_refill_mono_sec: now_mono,
                full_at_mono_sec: full_at,
                bucket_gen,
            },
        );
        self.reclaim_heap.push(IpReclaimEntry {
            full_at_mono_sec: full_at,
            ip,
            bucket_gen,
        });
        self.maintain_heap(now_mono, 0);
        true
    }

    #[cfg(test)]
    fn heap_len(&self) -> usize {
        self.reclaim_heap.len()
    }
}

struct GlobalRateLimiter {
    millitokens: u32,
    last_refill_mono_sec: u64,
    capacity_mt: u32,
    refill_mt_per_sec: u32,
}

impl GlobalRateLimiter {
    fn new(capacity_mt: u32, refill_mt_per_sec: u32, now_mono: u64) -> Self {
        Self {
            millitokens: capacity_mt,
            last_refill_mono_sec: now_mono,
            capacity_mt,
            refill_mt_per_sec: refill_mt_per_sec.max(1),
        }
    }

    fn refill(&mut self, now_mono: u64) {
        let elapsed = now_mono.saturating_sub(self.last_refill_mono_sec);
        let added = elapsed
            .saturating_mul(u64::from(self.refill_mt_per_sec))
            .min(u64::from(self.capacity_mt)) as u32;
        self.millitokens = self.millitokens.saturating_add(added).min(self.capacity_mt);
        self.last_refill_mono_sec = now_mono;
    }

    fn can_afford(&mut self, now_mono: u64, cost_mt: u32) -> bool {
        self.refill(now_mono);
        self.millitokens >= cost_mt
    }

    fn consume(&mut self, now_mono: u64, cost_mt: u32) -> bool {
        self.refill(now_mono);
        if self.millitokens < cost_mt {
            return false;
        }
        self.millitokens -= cost_mt;
        true
    }

    #[cfg(test)]
    fn available_millitokens(&mut self, now_mono: u64) -> u32 {
        self.refill(now_mono);
        self.millitokens
    }
}

/// Move-only proof of a durably committed pre-dispatch reservation.
pub struct DispatchPermit {
    scheduler: Arc<SamplerScheduler>,
    addr: SocketAddrV4,
    reservation_seq: u64,
    disarmed: bool,
}

impl fmt::Debug for DispatchPermit {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("DispatchPermit")
            .field("addr", &self.addr)
            .field("reservation_seq", &self.reservation_seq)
            .field("disarmed", &self.disarmed)
            .finish()
    }
}

impl DispatchPermit {
    pub fn addr(&self) -> SocketAddrV4 {
        self.addr
    }

    pub fn reservation_seq(&self) -> u64 {
        self.reservation_seq
    }
}

impl Drop for DispatchPermit {
    fn drop(&mut self) {
        if !self.disarmed {
            self.scheduler
                .reconcile_abandoned_in_flight(self.addr, self.reservation_seq);
        }
    }
}

struct AdmissionCancelGuard<'a> {
    scheduler: &'a SamplerScheduler,
    addr: SocketAddrV4,
    reservation_seq: u64,
    armed: bool,
}

impl Drop for AdmissionCancelGuard<'_> {
    fn drop(&mut self) {
        if self.armed {
            self.scheduler
                .reconcile_cancelled_admission(self.addr, self.reservation_seq);
        }
    }
}

struct CompletionCancelGuard<'a> {
    scheduler: &'a SamplerScheduler,
    addr: SocketAddrV4,
    completion_seq: u64,
    armed: bool,
}

impl Drop for CompletionCancelGuard<'_> {
    fn drop(&mut self) {
        if self.armed {
            self.scheduler
                .reconcile_cancelled_completion(self.addr, self.completion_seq);
        }
    }
}

impl SamplerScheduler {
    pub fn open(db_path: impl AsRef<Path>) -> Result<Arc<Self>, SchedulerError> {
        Self::open_with_config(
            db_path,
            SchedulerConfig::default(),
            Arc::new(SystemClock::new()),
        )
    }

    pub fn open_with_config(
        db_path: impl AsRef<Path>,
        config: SchedulerConfig,
        clock: Arc<dyn Clock>,
    ) -> Result<Arc<Self>, SchedulerError> {
        let mut conn = Connection::open(db_path.as_ref())?;
        conn.execute_batch(
            "PRAGMA journal_mode = WAL;
             PRAGMA synchronous = FULL;
             PRAGMA busy_timeout = 5000;
             CREATE TABLE IF NOT EXISTS bep51_cooldowns (
                 ip_u32 INTEGER NOT NULL,
                 port INTEGER NOT NULL,
                 next_allowed_sec INTEGER NOT NULL,
                 support INTEGER NOT NULL,
                 last_novel INTEGER NOT NULL,
                 op_seq INTEGER NOT NULL,
                 PRIMARY KEY (ip_u32, port)
             );
             CREATE TABLE IF NOT EXISTS scheduler_meta (
                 id INTEGER PRIMARY KEY CHECK (id = 1),
                 max_op_seq INTEGER NOT NULL,
                 last_mono_epoch_sec INTEGER NOT NULL,
                 last_wall_epoch_sec INTEGER NOT NULL
             );",
        )?;

        let current_wall_sec = clock.wall_unix_secs()?;
        let anchor_mono_sec = clock.monotonic_elapsed_secs();
        let metrics = Arc::new(SchedulerMetrics::default());

        // Read persisted metadata to prevent sequence regression and detect wall-clock jumps across restarts.
        let meta_row: Option<(i64, i64, i64)> = conn
            .query_row(
                "SELECT max_op_seq, last_mono_epoch_sec, last_wall_epoch_sec FROM scheduler_meta WHERE id = 1",
                [],
                |r| Ok((r.get(0)?, r.get(1)?, r.get(2)?)),
            )
            .ok();

        let (meta_max_seq, anchor_epoch_sec) = if let Some((seq_raw, mono_raw, wall_raw)) = meta_row
        {
            if seq_raw < 0 || mono_raw < 0 || wall_raw < 0 {
                return Err(SchedulerError::CorruptDatabaseRecord {
                    ip_raw: 0,
                    port_raw: 0,
                    support_raw: 0,
                    novel_raw: 0,
                    op_seq_raw: seq_raw,
                });
            }
            let persisted_seq = u64::try_from(seq_raw).unwrap_or(0);
            let persisted_mono = u32::try_from(mono_raw).unwrap_or(current_wall_sec);
            let persisted_wall = u32::try_from(wall_raw).unwrap_or(current_wall_sec);

            // Conservative restart clock verification:
            // If current wall clock went backward OR jumped forward by more than `max_trusted_restart_advance_secs`,
            // credit zero offline elapsed time and resume directly from `persisted_mono`.
            let trusted_epoch = if current_wall_sec < persisted_wall {
                metrics.clock_drift_warnings.fetch_add(1, Ordering::Relaxed);
                persisted_mono
            } else {
                let wall_delta = current_wall_sec - persisted_wall;
                if wall_delta > config.max_trusted_restart_advance_secs {
                    metrics.clock_drift_warnings.fetch_add(1, Ordering::Relaxed);
                    warn!(
                        wall_delta_secs = wall_delta,
                        max_trusted = config.max_trusted_restart_advance_secs,
                        "Detected unverified forward wall-clock jump across restart; preserving cooldowns at last monotonic checkpoint"
                    );
                    persisted_mono
                } else {
                    persisted_mono.saturating_add(wall_delta).max(current_wall_sec)
                }
            };
            (persisted_seq, trusted_epoch)
        } else {
            conn.execute(
                "INSERT INTO scheduler_meta (id, max_op_seq, last_mono_epoch_sec, last_wall_epoch_sec)
                 VALUES (1, 0, ?1, ?2)",
                params![current_wall_sec, current_wall_sec],
            )?;
            (0, current_wall_sec)
        };

        let mut ledger = HashMap::new();
        let mut hot_heap = BinaryHeap::new();
        let mut expiry_heap = BinaryHeap::new();
        let mut expired_to_delete: Vec<(SocketAddrV4, u64)> = Vec::new();
        let mut highest_op_seq: u64 = meta_max_seq;

        {
            let mut stmt = conn.prepare(
                "SELECT ip_u32, port, next_allowed_sec, support, last_novel, op_seq
                 FROM bep51_cooldowns",
            )?;
            let mut rows = stmt.query([])?;

            while let Some(row) = rows.next()? {
                let ip_raw: i64 = row.get(0)?;
                let port_raw: i64 = row.get(1)?;
                let next_raw: i64 = row.get(2)?;
                let support_raw: i64 = row.get(3)?;
                let novel_raw: i64 = row.get(4)?;
                let op_seq_raw: i64 = row.get(5)?;

                let corrupt = || SchedulerError::CorruptDatabaseRecord {
                    ip_raw,
                    port_raw,
                    support_raw,
                    novel_raw,
                    op_seq_raw,
                };

                let ip_u32 = u32::try_from(ip_raw).map_err(|_| corrupt())?;
                let port_u16 = u16::try_from(port_raw).map_err(|_| corrupt())?;
                let next_allowed_sec = u32::try_from(next_raw).map_err(|_| corrupt())?;
                let support = Bep51Support::from_db_i64(support_raw).ok_or_else(corrupt)?;
                let last_novel = u8::try_from(novel_raw).map_err(|_| corrupt())?;
                if op_seq_raw < 1 {
                    return Err(corrupt());
                }
                let op_seq = u64::try_from(op_seq_raw).map_err(|_| corrupt())?;
                highest_op_seq = highest_op_seq.max(op_seq);

                let addr = SocketAddrV4::new(Ipv4Addr::from(ip_u32), port_u16);
                if !is_valid_public_dht_endpoint(addr) {
                    return Err(corrupt());
                }

                let is_productive = support == Bep51Support::Supported && last_novel >= 2;
                if next_allowed_sec <= anchor_epoch_sec && !is_productive {
                    expired_to_delete.push((addr, op_seq));
                    continue;
                }

                // Check capacity after filtering expired non-productive rows.
                if ledger.len() >= config.max_cooldown_entries {
                    return Err(SchedulerError::DatabaseOverflow {
                        found: ledger.len() + 1,
                        max: config.max_cooldown_entries,
                    });
                }

                let rec = EndpointRecord {
                    next_allowed_epoch_sec: next_allowed_sec,
                    support,
                    last_novel_hashes: last_novel,
                    op_seq,
                    lifecycle: Lifecycle::Cooling,
                };
                ledger.insert(addr, rec);

                let node = ScheduledNode {
                    due_epoch_sec: next_allowed_sec,
                    addr,
                    op_seq,
                };
                if is_productive {
                    hot_heap.push(node);
                } else {
                    expiry_heap.push(node);
                }
            }
        }

        // Persist updated high-water op_seq and delete reclaimed expired rows before starting.
        {
            let tx = conn.transaction_with_behavior(TransactionBehavior::Immediate)?;
            tx.execute(
                "UPDATE scheduler_meta 
                 SET max_op_seq = MAX(max_op_seq, ?1),
                     last_mono_epoch_sec = ?2,
                     last_wall_epoch_sec = ?3
                 WHERE id = 1",
                params![highest_op_seq as i64, anchor_epoch_sec, current_wall_sec],
            )?;
            if !expired_to_delete.is_empty() {
                let mut del_stmt = tx.prepare_cached(
                    "DELETE FROM bep51_cooldowns WHERE ip_u32 = ?1 AND port = ?2 AND op_seq = ?3",
                )?;
                for (addr, seq) in expired_to_delete {
                    del_stmt.execute(params![u32::from(*addr.ip()), addr.port(), seq as i64])?;
                }
            }
            tx.commit()?;
        }

        info!(
            loaded_records = ledger.len(),
            loaded_hot_nodes = hot_heap.len(),
            highest_op_seq,
            "Initialized crash-safe BEP 51 SamplerScheduler"
        );

        let (wal_tx, wal_rx) = mpsc::channel::<WalWriteRequest>(4_096);
        let metrics_writer = Arc::clone(&metrics);
        let inject_wal_failure = Arc::new(AtomicBool::new(false));
        let inject_writer = Arc::clone(&inject_wal_failure);
        let batch_max_items = config.wal_batch_max_items;
        let batch_max_wait = config.wal_batch_max_wait;

        std::thread::Builder::new()
            .name("lodestar-bep51-wal".into())
            .spawn(move || {
                Self::run_group_commit_writer(
                    conn,
                    wal_rx,
                    metrics_writer,
                    inject_writer,
                    batch_max_items,
                    batch_max_wait,
                );
            })
            .expect("failed to spawn WAL writer thread");

        let ip_limiter = IpRateLimiter::new(
            config.max_ip_bucket_entries,
            config.ip_bucket_capacity_mt,
            config.ip_bucket_refill_mt_per_sec,
        );
        let global_limiter = GlobalRateLimiter::new(
            config.global_bucket_capacity_mt,
            config.global_bucket_refill_mt_per_sec,
            anchor_mono_sec,
        );

        Ok(Arc::new(Self {
            core: Mutex::new(SchedulerCore {
                ledger,
                ready_queue: VecDeque::with_capacity(config.max_ready_queue_len.min(16_384)),
                hot_heap,
                expiry_heap,
                ip_limiter,
                global_limiter,
                in_flight_count: 0,
                hot_dispatch_streak: 0,
                next_op_seq: highest_op_seq,
                anchor_epoch_sec,
                anchor_mono_sec,
            }),
            config,
            clock,
            wal_tx,
            inject_wal_failure,
            drop_reclaim_deletes: AtomicBool::new(false),
            metrics,
        }))
    }

    /// Evaluates current time strictly from the monotonic clock anchor (`mono_epoch`).
    /// Never lets a forward wall-clock jump advance `mono_epoch` or inflate written deadlines.
    fn sample_clocks(&self, core: &SchedulerCore) -> Result<(u32, u32, u64), SchedulerError> {
        let now_mono = self.clock.monotonic_elapsed_secs();
        let delta_mono = now_mono.saturating_sub(core.anchor_mono_sec);
        let mono_epoch = core
            .anchor_epoch_sec
            .saturating_add(u32::try_from(delta_mono).unwrap_or(u32::MAX));

        let wall_epoch = self.clock.wall_unix_secs()?;
        if wall_epoch.abs_diff(mono_epoch) > MAX_CLOCK_DRIFT_WARN_SECS {
            self.metrics
                .clock_drift_warnings
                .fetch_add(1, Ordering::Relaxed);
        }
        Ok((mono_epoch, wall_epoch, now_mono))
    }

    fn reclaim_expired_locked(&self, core: &mut SchedulerCore, mono_now: u32, max_reclaim: usize) {
        let mut reclaimed = 0;
        let suppress_delete = self.drop_reclaim_deletes.load(Ordering::Relaxed);

        while reclaimed < max_reclaim {
            let Some(&node) = core.expiry_heap.peek() else {
                break;
            };
            if node.due_epoch_sec > mono_now {
                break;
            }
            core.expiry_heap.pop();

            let can_remove = if let Some(rec) = core.ledger.get(&node.addr) {
                rec.op_seq == node.op_seq
                    && rec.lifecycle == Lifecycle::Cooling
                    && mono_now >= rec.next_allowed_epoch_sec
            } else {
                false
            };

            if can_remove {
                core.ledger.remove(&node.addr);
                reclaimed += 1;
                self.metrics
                    .expired_records_reclaimed
                    .fetch_add(1, Ordering::Relaxed);
                if !suppress_delete {
                    let _ = self.wal_tx.try_send(WalWriteRequest {
                        op: WalOperation::DeleteIfSeqMatches {
                            addr: node.addr,
                            expected_op_seq: node.op_seq,
                        },
                        ack_tx: None,
                    });
                }
            }
        }

        let heap_cap = self.config.max_cooldown_entries.saturating_mul(2).max(1_024);
        if core.expiry_heap.len() > heap_cap {
            let ledger_ref = &core.ledger;
            let filtered: Vec<ScheduledNode> = core
                .expiry_heap
                .drain()
                .filter(|n| {
                    ledger_ref
                        .get(&n.addr)
                        .map(|r| r.op_seq == n.op_seq && r.lifecycle == Lifecycle::Cooling)
                        .unwrap_or(false)
                })
                .collect();
            core.expiry_heap.extend(filtered);
        }
        if core.hot_heap.len() > heap_cap {
            let ledger_ref = &core.ledger;
            let filtered: Vec<ScheduledNode> = core
                .hot_heap
                .drain()
                .filter(|n| {
                    ledger_ref
                        .get(&n.addr)
                        .map(|r| r.op_seq == n.op_seq && r.lifecycle == Lifecycle::Cooling)
                        .unwrap_or(false)
                })
                .collect();
            core.hot_heap.extend(filtered);
        }
    }

    pub fn consider_candidate(&self, addr: SocketAddrV4) -> Result<bool, SchedulerError> {
        if !is_valid_public_dht_endpoint(addr) {
            self.metrics
                .rejected_bogon_or_private_ip
                .fetch_add(1, Ordering::Relaxed);
            return Ok(false);
        }

        let mut core = self.core.lock().expect("scheduler core mutex poisoned");
        let (mono_now, _, _) = self.sample_clocks(&core)?;

        self.reclaim_expired_locked(&mut core, mono_now, 8);

        if core.ready_queue.len() >= self.config.max_ready_queue_len {
            self.metrics
                .skipped_ready_queue_full
                .fetch_add(1, Ordering::Relaxed);
            return Ok(false);
        }

        if let Some(rec) = core.ledger.get_mut(&addr) {
            match rec.lifecycle {
                Lifecycle::Queued
                | Lifecycle::PendingWalReservation { .. }
                | Lifecycle::InFlight { .. }
                | Lifecycle::Completing { .. } => {
                    self.metrics
                        .skipped_already_active
                        .fetch_add(1, Ordering::Relaxed);
                    return Ok(false);
                }
                Lifecycle::Cooling => {
                    if rec.support == Bep51Support::Unsupported
                        && mono_now < rec.next_allowed_epoch_sec
                    {
                        self.metrics
                            .skipped_unsupported
                            .fetch_add(1, Ordering::Relaxed);
                        return Ok(false);
                    }
                    if mono_now < rec.next_allowed_epoch_sec {
                        self.metrics
                            .skipped_on_cooldown
                            .fetch_add(1, Ordering::Relaxed);
                        return Ok(false);
                    }
                    rec.lifecycle = Lifecycle::Queued;
                    core.ready_queue.push_back(addr);
                    self.metrics
                        .candidates_admitted_to_ready
                        .fetch_add(1, Ordering::Relaxed);
                    return Ok(true);
                }
            }
        }

        if core.ledger.len() >= self.config.max_cooldown_entries {
            self.reclaim_expired_locked(&mut core, mono_now, 64);
            if core.ledger.len() >= self.config.max_cooldown_entries {
                self.metrics
                    .skipped_ledger_full
                    .fetch_add(1, Ordering::Relaxed);
                return Ok(false);
            }
        }

        core.ledger.insert(
            addr,
            EndpointRecord {
                next_allowed_epoch_sec: mono_now,
                support: Bep51Support::Unknown,
                last_novel_hashes: 0,
                op_seq: 0,
                lifecycle: Lifecycle::Queued,
            },
        );
        core.ready_queue.push_back(addr);
        self.metrics
            .candidates_admitted_to_ready
            .fetch_add(1, Ordering::Relaxed);
        Ok(true)
    }

    fn try_pop_hot_locked(
        &self,
        core: &mut SchedulerCore,
        mono_now: u32,
        now_mono_sec: u64,
    ) -> Option<SocketAddrV4> {
        while let Some(&node) = core.hot_heap.peek() {
            if node.due_epoch_sec > mono_now {
                break;
            }
            core.hot_heap.pop();

            let is_eligible = if let Some(rec) = core.ledger.get(&node.addr) {
                rec.op_seq == node.op_seq
                    && rec.lifecycle == Lifecycle::Cooling
                    && mono_now >= rec.next_allowed_epoch_sec
            } else {
                false
            };

            if !is_eligible {
                continue;
            }

            if core
                .ip_limiter
                .try_consume(*node.addr.ip(), now_mono_sec, MILLITOKENS_PER_REQUEST)
            {
                self.metrics
                    .dispatched_from_hot_heap
                    .fetch_add(1, Ordering::Relaxed);
                return Some(node.addr);
            } else {
                self.metrics
                    .throttled_by_ip_bucket
                    .fetch_add(1, Ordering::Relaxed);
                core.hot_heap.push(ScheduledNode {
                    due_epoch_sec: mono_now.saturating_add(10),
                    addr: node.addr,
                    op_seq: node.op_seq,
                });
            }
        }
        None
    }

    fn try_pop_ready_locked(
        &self,
        core: &mut SchedulerCore,
        now_mono_sec: u64,
    ) -> Option<SocketAddrV4> {
        let queue_len = core.ready_queue.len();
        for _ in 0..queue_len {
            let Some(addr) = core.ready_queue.pop_front() else {
                break;
            };
            let is_queued = core
                .ledger
                .get(&addr)
                .map(|r| r.lifecycle == Lifecycle::Queued)
                .unwrap_or(false);

            if !is_queued {
                continue;
            }

            if core
                .ip_limiter
                .try_consume(*addr.ip(), now_mono_sec, MILLITOKENS_PER_REQUEST)
            {
                self.metrics
                    .dispatched_from_ready_queue
                    .fetch_add(1, Ordering::Relaxed);
                return Some(addr);
            } else {
                self.metrics
                    .throttled_by_ip_bucket
                    .fetch_add(1, Ordering::Relaxed);
                core.ready_queue.push_back(addr);
            }
        }
        None
    }

    /// Selects an eligible candidate, allocates a globally monotonic `op_seq`, and commits a 6-hour
    /// protective reservation to SQLite WAL (requiring `rows_affected == 1`) before returning a `DispatchPermit`.
    pub async fn acquire_next_dispatch(
        self: &Arc<Self>,
    ) -> Result<Option<DispatchPermit>, SchedulerError> {
        let (
            candidate_addr,
            reservation_seq,
            protective_sec,
            support,
            last_novel,
            mono_now,
            wall_now,
        ) = {
            let mut core = self.core.lock().expect("scheduler core mutex poisoned");
            let (mono_now, wall_now, now_mono_sec) = self.sample_clocks(&core)?;

            if core.in_flight_count >= self.config.max_in_flight_requests {
                self.metrics
                    .throttled_by_in_flight_cap
                    .fetch_add(1, Ordering::Relaxed);
                return Ok(None);
            }

            // Peek global budget WITHOUT consuming tokens if no candidate is eligible.
            if !core
                .global_limiter
                .can_afford(now_mono_sec, MILLITOKENS_PER_REQUEST)
            {
                self.metrics
                    .throttled_by_global_limit
                    .fetch_add(1, Ordering::Relaxed);
                return Ok(None);
            }

            let prefer_ready = core.hot_dispatch_streak >= self.config.max_hot_streak
                && !core.ready_queue.is_empty();

            let selected = if prefer_ready {
                if let Some(addr) = self.try_pop_ready_locked(&mut core, now_mono_sec) {
                    core.hot_dispatch_streak = 0;
                    Some(addr)
                } else if let Some(addr) =
                    self.try_pop_hot_locked(&mut core, mono_now, now_mono_sec)
                {
                    core.hot_dispatch_streak = core.hot_dispatch_streak.saturating_add(1);
                    Some(addr)
                } else {
                    None
                }
            } else if let Some(addr) = self.try_pop_hot_locked(&mut core, mono_now, now_mono_sec) {
                core.hot_dispatch_streak = core.hot_dispatch_streak.saturating_add(1);
                Some(addr)
            } else if let Some(addr) = self.try_pop_ready_locked(&mut core, now_mono_sec) {
                core.hot_dispatch_streak = 0;
                Some(addr)
            } else {
                None
            };

            let Some(addr) = selected else {
                return Ok(None);
            };

            // Candidate is confirmed eligible; consume the global token now.
            let consumed = core
                .global_limiter
                .consume(now_mono_sec, MILLITOKENS_PER_REQUEST);
            debug_assert!(consumed, "global limiter must succeed after can_afford");

            let next_global_seq = core
                .next_op_seq
                .checked_add(1)
                .filter(|&s| i64::try_from(s).is_ok())
                .ok_or(SchedulerError::SequenceOverflow)?;

            let protective_sec = mono_now.saturating_add(MAX_BEP51_INTERVAL_SECS);
            let rec = core
                .ledger
                .get_mut(&addr)
                .expect("selected endpoint must exist");

            // Ensure new op_seq is strictly greater than both global counter and any endpoint state.
            let assigned_seq = next_global_seq.max(rec.op_seq.saturating_add(1));
            if i64::try_from(assigned_seq).is_err() {
                return Err(SchedulerError::SequenceOverflow);
            }
            core.next_op_seq = assigned_seq;

            rec.op_seq = assigned_seq;
            rec.next_allowed_epoch_sec = protective_sec;
            rec.lifecycle = Lifecycle::PendingWalReservation {
                reservation_seq: assigned_seq,
            };
            core.in_flight_count += 1;

            (
                addr,
                assigned_seq,
                protective_sec,
                rec.support,
                rec.last_novel_hashes,
                mono_now,
                wall_now,
            )
        };

        let mut cancel_guard = AdmissionCancelGuard {
            scheduler: self.as_ref(),
            addr: candidate_addr,
            reservation_seq,
            armed: true,
        };

        let (ack_tx, ack_rx) = oneshot::channel();
        let req = WalWriteRequest {
            op: WalOperation::Upsert {
                addr: candidate_addr,
                next_allowed_epoch_sec: protective_sec,
                support,
                last_novel_hashes: last_novel,
                op_seq: reservation_seq,
                mono_epoch_sec: mono_now,
                wall_epoch_sec: wall_now,
            },
            ack_tx: Some(ack_tx),
        };

        if self.wal_tx.send(req).await.is_err() {
            return Err(SchedulerError::WalChannelClosed);
        }

        match ack_rx.await {
            Ok(Ok(())) => {
                let mut core = self.core.lock().expect("scheduler core mutex poisoned");
                if let Some(rec) = core.ledger.get_mut(&candidate_addr) {
                    if rec.op_seq == reservation_seq
                        && rec.lifecycle == (Lifecycle::PendingWalReservation { reservation_seq })
                    {
                        rec.lifecycle = Lifecycle::InFlight { reservation_seq };
                        cancel_guard.armed = false;
                        self.metrics
                            .reservations_committed
                            .fetch_add(1, Ordering::Relaxed);
                        return Ok(Some(DispatchPermit {
                            scheduler: Arc::clone(self),
                            addr: candidate_addr,
                            reservation_seq,
                            disarmed: false,
                        }));
                    }
                }
                Err(SchedulerError::InvalidLifecycleTransition)
            }
            Ok(Err(e)) => Err(e),
            Err(_) => Err(SchedulerError::WalCommitAckDropped),
        }
    }

    fn push_cooling_heap_locked(core: &mut SchedulerCore, addr: SocketAddrV4, rec: &EndpointRecord) {
        let node = ScheduledNode {
            due_epoch_sec: rec.next_allowed_epoch_sec,
            addr,
            op_seq: rec.op_seq,
        };
        if rec.support == Bep51Support::Supported && rec.last_novel_hashes >= 2 {
            core.hot_heap.push(node);
        } else {
            core.expiry_heap.push(node);
        }
    }

    fn reconcile_cancelled_admission(&self, addr: SocketAddrV4, reservation_seq: u64) {
        let mut core = self.core.lock().expect("scheduler core mutex poisoned");
        core.in_flight_count = core.in_flight_count.saturating_sub(1);
        let snap = if let Some(rec) = core.ledger.get_mut(&addr) {
            if rec.op_seq == reservation_seq
                && rec.lifecycle == (Lifecycle::PendingWalReservation { reservation_seq })
            {
                rec.lifecycle = Lifecycle::Cooling;
                Some(*rec)
            } else {
                None
            }
        } else {
            None
        };
        if let Some(rec) = snap {
            Self::push_cooling_heap_locked(&mut core, addr, &rec);
            self.metrics
                .reservations_failed_or_cancelled
                .fetch_add(1, Ordering::Relaxed);
        }
    }

    fn reconcile_abandoned_in_flight(&self, addr: SocketAddrV4, reservation_seq: u64) {
        let mut core = self.core.lock().expect("scheduler core mutex poisoned");
        core.in_flight_count = core.in_flight_count.saturating_sub(1);
        let snap = if let Some(rec) = core.ledger.get_mut(&addr) {
            if rec.op_seq == reservation_seq
                && rec.lifecycle == (Lifecycle::InFlight { reservation_seq })
            {
                rec.lifecycle = Lifecycle::Cooling;
                Some(*rec)
            } else {
                None
            }
        } else {
            None
        };
        if let Some(rec) = snap {
            Self::push_cooling_heap_locked(&mut core, addr, &rec);
            self.metrics
                .abandoned_permits_reconciled
                .fetch_add(1, Ordering::Relaxed);
        }
    }

    fn reconcile_cancelled_completion(&self, addr: SocketAddrV4, completion_seq: u64) {
        let mut core = self.core.lock().expect("scheduler core mutex poisoned");
        core.in_flight_count = core.in_flight_count.saturating_sub(1);
        let snap = if let Some(rec) = core.ledger.get_mut(&addr) {
            if rec.op_seq == completion_seq
                && rec.lifecycle == (Lifecycle::Completing { completion_seq })
            {
                rec.lifecycle = Lifecycle::Cooling;
                Some(*rec)
            } else {
                None
            }
        } else {
            None
        };
        if let Some(rec) = snap {
            Self::push_cooling_heap_locked(&mut core, addr, &rec);
        }
    }

    /// Consumes the linear `DispatchPermit` and commits the outcome to SQLite WAL.
    /// Does NOT disarm `permit` until after the mutex is acquired and ownership is atomically
    /// transferred to `CompletionCancelGuard`, eliminating any early-return leak window.
    pub async fn complete_dispatch(
        &self,
        mut permit: DispatchPermit,
        outcome: SampleOutcome,
    ) -> Result<(), SchedulerError> {
        let addr = permit.addr;
        let reservation_seq = permit.reservation_seq;
        self.complete_dispatch_inner(addr, reservation_seq, Some(&mut permit), outcome)
            .await
    }

    pub(crate) async fn complete_dispatch_inner(
        &self,
        addr: SocketAddrV4,
        reservation_seq: u64,
        permit_opt: Option<&mut DispatchPermit>,
        outcome: SampleOutcome,
    ) -> Result<(), SchedulerError> {
        let (completion_seq, target_deadline, new_support, new_novel, mono_now, wall_now) = {
            let mut core = self.core.lock().expect("scheduler core mutex poisoned");
            let (mono_now, wall_now, _) = self.sample_clocks(&core)?;

            let next_global_seq = core
                .next_op_seq
                .checked_add(1)
                .filter(|&s| i64::try_from(s).is_ok())
                .ok_or(SchedulerError::SequenceOverflow)?;

            let Some(rec) = core.ledger.get_mut(&addr) else {
                return Err(SchedulerError::InvalidLifecycleTransition);
            };

            if rec.op_seq != reservation_seq {
                self.metrics
                    .stale_or_duplicate_completions_rejected
                    .fetch_add(1, Ordering::Relaxed);
                warn!(
                    endpoint = %addr,
                    expected = rec.op_seq,
                    provided = reservation_seq,
                    "Rejected stale or duplicate BEP 51 completion"
                );
                return Err(SchedulerError::StaleSequence {
                    expected: rec.op_seq,
                    provided: reservation_seq,
                });
            }

            if rec.lifecycle != (Lifecycle::InFlight { reservation_seq }) {
                self.metrics
                    .stale_or_duplicate_completions_rejected
                    .fetch_add(1, Ordering::Relaxed);
                return Err(SchedulerError::InvalidLifecycleTransition);
            }

            let completion_seq = next_global_seq.max(rec.op_seq.saturating_add(1));
            if i64::try_from(completion_seq).is_err() {
                return Err(SchedulerError::SequenceOverflow);
            }

            let (deadline, support, novel) = match outcome {
                SampleOutcome::Supported {
                    advertised_interval_sec,
                    novel_hashes_count,
                } => {
                    let raw = advertised_interval_sec.unwrap_or(MAX_BEP51_INTERVAL_SECS);
                    let delay = raw.clamp(MIN_POLITE_INTERVAL_FLOOR_SECS, MAX_BEP51_INTERVAL_SECS);
                    (
                        mono_now.saturating_add(delay),
                        Bep51Support::Supported,
                        novel_hashes_count,
                    )
                }
                SampleOutcome::ExplicitUnsupported => (
                    mono_now.saturating_add(UNSUPPORTED_BACKOFF_SECS),
                    Bep51Support::Unsupported,
                    0,
                ),
                SampleOutcome::Timeout => {
                    let preserved_support = if rec.support == Bep51Support::Supported {
                        Bep51Support::Supported
                    } else {
                        Bep51Support::TimeoutBackoff
                    };
                    (
                        mono_now.saturating_add(DEFAULT_TIMEOUT_BACKOFF_SECS),
                        preserved_support,
                        rec.last_novel_hashes,
                    )
                }
                SampleOutcome::Malformed => (
                    mono_now.saturating_add(MALFORMED_BACKOFF_SECS),
                    Bep51Support::MalformedBackoff,
                    0,
                ),
            };

            // Atomically transition to Completing and disarm the caller's permit inside the lock.
            core.next_op_seq = completion_seq;
            rec.op_seq = completion_seq;
            rec.lifecycle = Lifecycle::Completing { completion_seq };
            if let Some(p) = permit_opt {
                p.disarmed = true;
            }

            (completion_seq, deadline, support, novel, mono_now, wall_now)
        };

        let mut cancel_guard = CompletionCancelGuard {
            scheduler: self,
            addr,
            completion_seq,
            armed: true,
        };

        let (ack_tx, ack_rx) = oneshot::channel();
        let req = WalWriteRequest {
            op: WalOperation::Upsert {
                addr,
                next_allowed_epoch_sec: target_deadline,
                support: new_support,
                last_novel_hashes: new_novel,
                op_seq: completion_seq,
                mono_epoch_sec: mono_now,
                wall_epoch_sec: wall_now,
            },
            ack_tx: Some(ack_tx),
        };

        if self.wal_tx.send(req).await.is_err() {
            return Err(SchedulerError::WalChannelClosed);
        }

        match ack_rx.await {
            Ok(Ok(())) => {
                let mut core = self.core.lock().expect("scheduler core mutex poisoned");
                core.in_flight_count = core.in_flight_count.saturating_sub(1);
                let snap = if let Some(rec) = core.ledger.get_mut(&addr) {
                    if rec.op_seq == completion_seq
                        && rec.lifecycle == (Lifecycle::Completing { completion_seq })
                    {
                        rec.next_allowed_epoch_sec = target_deadline;
                        rec.support = new_support;
                        rec.last_novel_hashes = new_novel;
                        rec.lifecycle = Lifecycle::Cooling;
                        Some(*rec)
                    } else {
                        None
                    }
                } else {
                    None
                };

                if let Some(rec) = snap {
                    Self::push_cooling_heap_locked(&mut core, addr, &rec);
                    cancel_guard.armed = false;
                    self.metrics
                        .outcomes_committed
                        .fetch_add(1, Ordering::Relaxed);
                    return Ok(());
                }
                Err(SchedulerError::InvalidLifecycleTransition)
            }
            Ok(Err(e)) => Err(e),
            Err(_) => Err(SchedulerError::WalCommitAckDropped),
        }
    }

    pub fn try_acquire_ip_budget(&self, ip: Ipv4Addr, cost_mt: u32) -> Result<bool, SchedulerError> {
        if cost_mt == 0 {
            return Ok(false);
        }
        let mut core = self.core.lock().expect("scheduler core mutex poisoned");
        let (_, _, now_mono_sec) = self.sample_clocks(&core)?;

        if !core.global_limiter.can_afford(now_mono_sec, cost_mt) {
            self.metrics
                .throttled_by_global_limit
                .fetch_add(1, Ordering::Relaxed);
            return Ok(false);
        }
        if !core.ip_limiter.try_consume(ip, now_mono_sec, cost_mt) {
            self.metrics
                .throttled_by_ip_bucket
                .fetch_add(1, Ordering::Relaxed);
            return Ok(false);
        }
        let consumed = core.global_limiter.consume(now_mono_sec, cost_mt);
        debug_assert!(consumed);
        Ok(true)
    }

    pub fn set_inject_wal_failure(&self, fail: bool) {
        self.inject_wal_failure.store(fail, Ordering::SeqCst);
    }

    #[cfg(test)]
    fn set_drop_reclaim_deletes(&self, drop_deletes: bool) {
        self.drop_reclaim_deletes
            .store(drop_deletes, Ordering::SeqCst);
    }

    fn run_group_commit_writer(
        mut conn: Connection,
        mut rx: mpsc::Receiver<WalWriteRequest>,
        metrics: Arc<SchedulerMetrics>,
        inject_failure: Arc<AtomicBool>,
        batch_max_items: usize,
        batch_max_wait: Duration,
    ) {
        let mut batch: Vec<WalWriteRequest> = Vec::with_capacity(batch_max_items);

        while let Some(first) = rx.blocking_recv() {
            batch.push(first);
            let deadline = Instant::now() + batch_max_wait;

            while batch.len() < batch_max_items {
                match rx.try_recv() {
                    Ok(item) => batch.push(item),
                    Err(mpsc::error::TryRecvError::Empty) => {
                        if Instant::now() >= deadline {
                            break;
                        }
                        std::thread::sleep(Duration::from_millis(1));
                    }
                    Err(mpsc::error::TryRecvError::Disconnected) => break,
                }
            }

            let commit_result: Result<(), SchedulerError> = (|| {
                let tx = conn.transaction_with_behavior(TransactionBehavior::Immediate)?;
                if inject_failure.load(Ordering::SeqCst) {
                    return Err(SchedulerError::InjectedWalFailure);
                }

                let mut batch_max_seq: u64 = 0;
                let mut batch_mono_sec: u32 = 0;
                let mut batch_wall_sec: u32 = 0;

                {
                    let mut upsert_stmt = tx.prepare_cached(
                        "INSERT INTO bep51_cooldowns 
                         (ip_u32, port, next_allowed_sec, support, last_novel, op_seq)
                         VALUES (?1, ?2, ?3, ?4, ?5, ?6)
                         ON CONFLICT(ip_u32, port) DO UPDATE SET
                             next_allowed_sec = excluded.next_allowed_sec,
                             support = excluded.support,
                             last_novel = excluded.last_novel,
                             op_seq = excluded.op_seq
                         WHERE excluded.op_seq > bep51_cooldowns.op_seq",
                    )?;

                    let mut del_stmt = tx.prepare_cached(
                        "DELETE FROM bep51_cooldowns 
                         WHERE ip_u32 = ?1 AND port = ?2 AND op_seq = ?3",
                    )?;

                    for req in &batch {
                        match req.op {
                            WalOperation::Upsert {
                                addr,
                                next_allowed_epoch_sec,
                                support,
                                last_novel_hashes,
                                op_seq,
                                mono_epoch_sec,
                                wall_epoch_sec,
                            } => {
                                let ip_u32 = u32::from(*addr.ip());
                                let changed = upsert_stmt.execute(params![
                                    ip_u32,
                                    addr.port(),
                                    next_allowed_epoch_sec,
                                    support as u8,
                                    last_novel_hashes,
                                    op_seq as i64,
                                ])?;

                                // Mandatory proof of persistence: if SQLite updated 0 rows, fail closed!
                                if changed != 1 {
                                    return Err(SchedulerError::ZeroRowsAffected { addr, op_seq });
                                }
                                batch_max_seq = batch_max_seq.max(op_seq);
                                batch_mono_sec = batch_mono_sec.max(mono_epoch_sec);
                                batch_wall_sec = batch_wall_sec.max(wall_epoch_sec);
                            }
                            WalOperation::DeleteIfSeqMatches {
                                addr,
                                expected_op_seq,
                            } => {
                                let ip_u32 = u32::from(*addr.ip());
                                del_stmt.execute(params![
                                    ip_u32,
                                    addr.port(),
                                    expected_op_seq as i64,
                                ])?;
                            }
                        }
                    }

                    if batch_max_seq > 0 {
                        tx.execute(
                            "UPDATE scheduler_meta
                             SET max_op_seq = MAX(max_op_seq, ?1),
                                 last_mono_epoch_sec = MAX(last_mono_epoch_sec, ?2),
                                 last_wall_epoch_sec = ?3
                             WHERE id = 1",
                            params![batch_max_seq as i64, batch_mono_sec, batch_wall_sec],
                        )?;
                    }
                }
                tx.commit()?;
                Ok(())
            })();

            match commit_result {
                Ok(()) => {
                    metrics
                        .wal_batches_committed
                        .fetch_add(1, Ordering::Relaxed);
                    for req in batch.drain(..) {
                        if let Some(ack_tx) = req.ack_tx {
                            let _ = ack_tx.send(Ok(()));
                        }
                    }
                }
                Err(err) => {
                    metrics.wal_commit_failures.fetch_add(1, Ordering::Relaxed);
                    error!(
                        batch_size = batch.len(),
                        error = %err,
                        "SQLite WAL group-commit failed; rolling back and notifying callers to fail closed"
                    );
                    for req in batch.drain(..) {
                        if let Some(ack_tx) = req.ack_tx {
                            let _ = ack_tx.send(Err(err.clone()));
                        }
                    }
                }
            }
        }
    }
}

    struct MockClock {
        wall_sec: AtomicU32,
        mono_sec: AtomicU64,
    }

    impl MockClock {
        fn new(initial_wall: u32, initial_mono: u64) -> Arc<Self> {
            Arc::new(Self {
                wall_sec: AtomicU32::new(initial_wall),
                mono_sec: AtomicU64::new(initial_mono),
            })
        }

        fn advance_both(&self, secs: u32) {
            self.wall_sec.fetch_add(secs, Ordering::SeqCst);
            self.mono_sec.fetch_add(u64::from(secs), Ordering::SeqCst);
        }

        fn jump_wall_only(&self, new_wall: u32) {
            self.wall_sec.store(new_wall, Ordering::SeqCst);
        }
    }

    impl Clock for MockClock {
        fn wall_unix_secs(&self) -> Result<u32, SchedulerError> {
            Ok(self.wall_sec.load(Ordering::SeqCst))
        }

        fn monotonic_elapsed_secs(&self) -> u64 {
            self.mono_sec.load(Ordering::SeqCst)
        }
    }

    /// Release-Gate Test 1: Drop an expired-record SQLite deletion, then rediscover the endpoint.
    /// Verifies that `next_op_seq` advances globally so the new 6-hour protective reservation
    /// overwrites the stale SQLite row (`rows_affected == 1`) and survives restart.
    #[tokio::test]
    async fn test_rg1_dropped_delete_then_rediscovery_persists_reservation() {
        let tmp = NamedTempFile::new().expect("temp db");
        let clock = MockClock::new(1_700_000_000, 100);
        let cfg = SchedulerConfig {
            max_cooldown_entries: 1,
            ..SchedulerConfig::default()
        };
        let sched =
            SamplerScheduler::open_with_config(tmp.path(), cfg.clone(), clock.clone()).unwrap();
        sched.set_drop_reclaim_deletes(true); // Simulate dropped WAL delete!

        let addr_a = SocketAddrV4::new(Ipv4Addr::new(93, 184, 216, 34), 6881);
        let addr_b = SocketAddrV4::new(Ipv4Addr::new(8, 8, 8, 8), 6881);

        // 1. Sample addr_a (reservation op_seq=1, completion op_seq=2 with 60s cooldown).
        assert!(sched.consider_candidate(addr_a).unwrap());
        let p1 = sched.acquire_next_dispatch().await.unwrap().unwrap();
        assert_eq!(p1.reservation_seq(), 1);
        sched
            .complete_dispatch(
                p1,
                SampleOutcome::Supported {
                    advertised_interval_sec: Some(60),
                    novel_hashes_count: 0,
                },
            )
            .await
            .unwrap();

        // 2. Advance 61s so addr_a expires; admit addr_b to force reclamation of addr_a from RAM while its SQLite delete is dropped!
        clock.advance_both(61);
        assert!(sched.consider_candidate(addr_b).unwrap());
        let pb = sched.acquire_next_dispatch().await.unwrap().unwrap();
        sched
            .complete_dispatch(
                pb,
                SampleOutcome::Supported {
                    advertised_interval_sec: Some(60),
                    novel_hashes_count: 0,
                },
            )
            .await
            .unwrap();

        // 3. Advance another 61s and rediscover addr_a (SQLite still has addr_a with op_seq=2 and expired deadline!).
        clock.advance_both(61);
        assert!(sched.consider_candidate(addr_a).unwrap());
        let p2 = sched.acquire_next_dispatch().await.unwrap().unwrap();
        assert!(
            p2.reservation_seq() > 2,
            "Rediscovered endpoint must receive globally higher op_seq ({}) > 2",
            p2.reservation_seq()
        );

        // Verify SQLite actually persisted the new 6-hour protective reservation before completion.
        let conn = Connection::open(tmp.path()).unwrap();
        let (stored_deadline, stored_seq): (u32, i64) = conn
            .query_row(
                "SELECT next_allowed_sec, op_seq FROM bep51_cooldowns WHERE ip_u32 = ?1 AND port = ?2",
                params![u32::from(*addr_a.ip()), addr_a.port()],
                |r| Ok((r.get(0)?, r.get(1)?)),
            )
            .unwrap();
        assert_eq!(stored_seq as u64, p2.reservation_seq());
        assert!(stored_deadline >= 1_700_000_122 + MAX_BEP51_INTERVAL_SECS);
    }

                SamplerScheduler::open_with_config(tmp.path(), cfg.clone(), clock.clone()).unwrap();
            let addr = SocketAddrV4::new(Ipv4Addr::new(8, 8, 8, 8), 6881);
            assert!(sched.consider_candidate(addr).unwrap());
            let p = sched.acquire_next_dispatch().await.unwrap().unwrap();
            sched
                .complete_dispatch(
                    p,
                    SampleOutcome::Supported {
                        advertised_interval_sec: Some(60),
                        novel_hashes_count: 0,
                    },
                )
                .await
                .unwrap();
        }

        // Insert a second already-expired row in SQLite so total rows (2) > max_cooldown_entries (1),
        // while active unexpired rows == 0 after advancing 61 seconds.
        {
            let conn = Connection::open(tmp.path()).unwrap();
            conn.execute(
                "INSERT INTO bep51_cooldowns (ip_u32, port, next_allowed_sec, support, last_novel, op_seq)
                 VALUES (?1, 6881, ?2, 1, 0, 10)",
                params![u32::from(Ipv4Addr::new(1, 1, 1, 1)), 1_700_000_010_u32],
            )
            .unwrap();
        }

        clock.advance_both(61);
        let reopened = SamplerScheduler::open_with_config(tmp.path(), cfg, clock);
        assert!(
            reopened.is_ok(),
            "Startup must reclaim expired non-productive rows before failing on max_cooldown_entries"
        );
    }

    /// Release-Gate Test 9: Concurrent completions for the same permit admit at most one.
    #[tokio::test]
    async fn test_rg9_concurrent_completions_same_permit_admits_exactly_one() {
        let tmp = NamedTempFile::new().expect("temp db");
        let clock = MockClock::new(1_700_000_000, 100);
        let sched =
            SamplerScheduler::open_with_config(tmp.path(), SchedulerConfig::default(), clock)
                .unwrap();

        let addr = SocketAddrV4::new(Ipv4Addr::new(8, 8, 8, 8), 6881);
        assert!(sched.consider_candidate(addr).unwrap());

        let mut permit = sched.acquire_next_dispatch().await.unwrap().unwrap();
        let res_seq = permit.reservation_seq();
        permit.disarmed = true;

        let s1 = Arc::clone(&sched);
        let s2 = Arc::clone(&sched);

        let (r1, r2) = tokio::join!(
            s1.complete_dispatch_inner(
                addr,
                res_seq,
                None,
                SampleOutcome::Supported {
                    advertised_interval_sec: Some(120),
                    novel_hashes_count: 4,
                }
            ),
            s2.complete_dispatch_inner(addr, res_seq, None, SampleOutcome::ExplicitUnsupported)
        );

        assert_eq!(usize::from(r1.is_ok()) + usize::from(r2.is_ok()), 1);
        assert_eq!(usize::from(r1.is_err()) + usize::from(r2.is_err()), 1);
    }

    /// Release-Gate Test 10: Subprocess abrupt `abort()` preserves the 6-hour protective reservation.
    #[tokio::test]
    async fn test_rg10_subprocess_abrupt_abort_preserves_6h_reservation() {
        let child_db_var = "LODESTAR_CRASH_TEST_DB_PATH";
        if let Ok(db_path) = std::env::var(child_db_var) {
            let sched = SamplerScheduler::open(&db_path).expect("child open");
            let addr = SocketAddrV4::new(Ipv4Addr::new(93, 184, 216, 34), 6881);
            assert!(sched.consider_candidate(addr).expect("child admit"));
            let _permit = sched
                .acquire_next_dispatch()
                .await
                .expect("child dispatch")
                .expect("child permit");
            std::process::abort();
        }

        let tmp = NamedTempFile::new().expect("temp db");
        let exe = std::env::current_exe().expect("current test binary");
        let status = std::process::Command::new(exe)
            .env(child_db_var, tmp.path())
            .arg("--exact")
            .arg("scheduler::tests::test_rg10_subprocess_abrupt_abort_preserves_6h_reservation")
            .arg("--nocapture")
            .status()
            .expect("spawn child process");

        assert!(!status.success());

        let restarted = SamplerScheduler::open(tmp.path()).expect("parent reopen after crash");
        let addr = SocketAddrV4::new(Ipv4Addr::new(93, 184, 216, 34), 6881);
        assert!(!restarted.consider_candidate(addr).expect("check after crash"));
    }
